# تصميم قاعدة البيانات: السوبرماركت في إب

## نظام إدارة قواعد البيانات
MySQL أو PostgreSQL مع دعم التوزيع الجغرافي.

## الجداول الرئيسية

### جدول المنتجات (products)
- id (PK, INT, AUTO_INCREMENT)
- name (VARCHAR(255))
- category_id (FK, INT)
- barcode (VARCHAR(50), UNIQUE)
- price (DECIMAL(10,2))
- cost_price (DECIMAL(10,2))
- quantity_in_stock (INT)
- expiry_date (DATE)
- image_url (TEXT)
- created_at (TIMESTAMP)

### جدول الفئات (categories)
- id (PK, INT)
- name (VARCHAR(100))
- parent_id (FK, INT, NULLABLE)
- icon (VARCHAR(255))

### جدول العملاء (customers)
- id (PK, INT)
- name (VARCHAR(100))
- phone (VARCHAR(20), UNIQUE)
- address (TEXT)
- loyalty_points (INT)
- created_at (TIMESTAMP)

### جدول المبيعات (sales)
- id (PK, INT)
- customer_id (FK, INT, NULLABLE)
- total_amount (DECIMAL(12,2))
- payment_method (ENUM)
- sale_date (TIMESTAMP)
- employee_id (FK, INT)

### جدول تفاصيل المبيعات (sale_items)
- id (PK, INT)
- sale_id (FK, INT)
- product_id (FK, INT)
- quantity (INT)
- unit_price (DECIMAL(10,2))
- subtotal (DECIMAL(12,2))

### جدول الموردين (suppliers)
- id (PK, INT)
- name (VARCHAR(150))
- phone (VARCHAR(20))
- address (TEXT)
- contract_start (DATE)
- contract_end (DATE)
